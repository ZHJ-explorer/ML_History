    class LinearRegression:
    """线性回归模型。

    使用最小二乘法求解，支持解析解和梯度下降两种方法。
    """

    def __init__(self, learning_rate=0.01, max_iter=1000, tol=1e-6):
        """初始化线性回归模型。

        Args:
            learning_rate: 学习率（梯度下降使用）。
            max_iter: 最大迭代次数（梯度下降使用）。
            tol: 收敛容差。
        """
        self.learning_rate = learning_rate
        self.max_iter = max_iter
        self.tol = tol
        self.weights = None
        self.bias = None
        self.history = None

    def _fit_closed_form(self, X, y):
        """使用正规方程求解。

        解析解：w = (X^T X)^(-1) X^T y

        Args:
            X: 特征矩阵，添加偏置列。
            y: 目标变量。
        """
        XTX = X.T @ X

        if np.linalg.det(XTX) == 0:
            self.weights = np.linalg.pinv(XTX) @ X.T @ y
        else:
            self.weights = np.linalg.inv(XTX) @ X.T @ y

        self.bias = self.weights[0]
        self.weights = self.weights[1:]

    X = np.random.rand(100, 1) * 10

    model = LinearRegression()
    model.fit(X, y)
    r2 = model.score(X, y)
    print(f"真实关系: y = 3x + 2")
    print(f"学习到的关系: y = {model.weights[0]:.3f}x + {model.bias:.3f}")
    print(f"R²分数: {r2:.4f}")

    局限性
    为什么它不够好

    1. **只能捕捉线性关系**：无法处理非线性问题

    2. **对异常值敏感**：MSE损失函数会被异常值影响

    3. **假设特征独立**：忽略了特征之间的相关性

    4. **需要特征缩放**：不同量纲的特征会影响收敛速度

    历史影响
    启发了什么

    线性回归直接启发了：

  • → 逻辑回归（二分类）
  • → 岭回归和Lasso回归（正则化）
  • → 广义线性模型
  • → 后续所有回归方法的理论基础

    从实践中学习

    学到的知识

      1. **最小二乘法**：线性回归的标准解法

      2. **两种求解方法**：解析解（正规方程）和梯度下降

      3. **特征缩放的重要性**：不同量纲的特征会影响收敛速度

      4. **R²分数**：衡量回归模型性能的标准指标

    思考题答案

      1. **为什么需要特征缩放？**

    • → 不同量纲的特征会导致梯度下降收敛缓慢
    • → 解析解不受影响，但数值稳定性可能受影响

      • 2. **正则化如何改善线性回归？**

    • → 岭回归（L2）：防止过拟合，处理共线性
    • → Lasso回归（L1）：特征选择，产生稀疏解

      • 3. **岭回归和Lasso回归有什么区别？**

    • → 岭回归：L2正则化，权重缩小但不为零
    • → Lasso回归：L1正则化，可以将某些权重压缩为零
