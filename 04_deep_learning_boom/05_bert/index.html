class BERT(nn.Module):
    def __init__(self, vocab_size=30522, d_model=768,
                 num_heads=12, num_layers=12):
        self.word_embedding = nn.Embedding(vocab_size, d_model)
        self.token_type_emb = nn.Embedding(2, d_model)
        self.position_emb = nn.Embedding(512, d_model)
        self.encoder = nn.TransformerEncoder(...)
        self.mlm_head = nn.Linear(d_model, vocab_size)
        self.nsp_head = nn.Linear(d_model, 2)

    def forward(self, input_ids, token_type_ids):
        x = self.word_embedding(input_ids) + \
            self.position_emb(position_ids) + \
            self.token_type_emb(token_type_ids)
        encoded = self.encoder(x)
        return self.mlm_head(encoded), self.nsp_head(encoded[:,0])
